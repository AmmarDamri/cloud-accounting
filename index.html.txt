    <script>
        let transactions = JSON.parse(localStorage.getItem('transactions')) || [];

        function updateUI() {
            const list = document.getElementById('transactionList');
            const netBalanceEl = document.getElementById('netBalance');
            const totalIncomeEl = document.getElementById('totalIncome');
            const totalExpenseEl = document.getElementById('totalExpense');

            list.innerHTML = '';
            let income = 0;
            let expense = 0;

            transactions.forEach((t, index) => {
                if (t.type === 'income') {
                    income += Number(t.amount);
                } else {
                    expense += Number(t.amount);
                }

                const row = document.createElement('tr');
                row.innerHTML = `
                    <td class="py-3 text-slate-700 font-medium">${t.desc}</td>
                    <td class="py-3">
                        <span class="px-2 py-1 text-xs rounded-full ${t.type === 'income' ? 'bg-emerald-50 text-emerald-600' : 'bg-rose-50 text-rose-600'}">
                            ${t.type.toUpperCase()}
                        </span>
                    </td>
                    <td class="py-3 text-right font-semibold ${t.type === 'income' ? 'text-emerald-600' : 'text-rose-600'}">
                        ${t.type === 'income' ? '+' : '-'} Rs. ${Number(t.amount).toLocaleString()}
                    </td>
                    <td class="py-3 text-center">
                        <button onclick="deleteTransaction(${index})" class="text-slate-400 hover:text-rose-600 text-xs">Delete</button>
                    </td>
                `;
                list.appendChild(row);
            });

            const net = income - expense;
            netBalanceEl.innerText = `Rs. ${net.toLocaleString()}`;
            totalIncomeEl.innerText = `Rs. ${income.toLocaleString()}`;
            totalExpenseEl.innerText = `Rs. ${expense.toLocaleString()}`;

            localStorage.setItem('transactions', JSON.stringify(transactions));
        }

        function addTransaction(e) {
            e.preventDefault();
            const desc = document.getElementById('desc').value;
            const amount = document.getElementById('amount').value;
            const type = document.getElementById('type').value;

            transactions.push({ desc, amount, type });
            document.getElementById('transactionForm').reset();
            updateUI();
        }

        function deleteTransaction(index) {
            transactions.splice(index, 1);
            updateUI();
        }

        // Initial Load
        updateUI();
    </script>
